1. Text Preprocessing
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
import nltk
nltk.download('punkt')
nltk.download('stopwords')
text = input("Enter text: ").lower()
tokens = word_tokenize(text)
stop_words = set(stopwords.words('english'))
filtered = [word for word in tokens if word.isalpha() and word not in stop_words]
print("Original Text:", text)
print("Tokens:", tokens)
print("Filtered Words:", filtered)

2. TF-IDF
import math
docs = [
  "The cat sat on the mat",
  "The dog played in the park",
  "Cat and dogs are great pets"
]
term = input("Enter term: ").lower()
N = len(docs)
df = sum(term in doc.lower().split() for doc in docs)
idf = math.log10(N / df) if df > 0 else 0
print("\nTF-IDF Calculation")
for i, doc in enumerate(docs, 1):
  words = doc.lower().split()
  tf = words.count(term) / len(words)
  tfidf = tf * idf
  print(f"Document {i}")
  print("TF =", round(tf, 3))
  print("IDF =", round(idf, 3))
  print("TF-IDF =", round(tfidf, 3))

3. Boolean Retrieval Model
docs = {
  1: "cat sat on mat",
  2: "dog played in park",
  3: "cat and dog are friends"
}
term = input("Enter search word: ").lower()
found = False
for doc_id, text in docs.items():
  if term in text.split():
    print("Term found in Document", doc_id)
    found = True
if not found:
  print("Term not found")

4. Vector Space Model (Cosine Similarity)
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
docs = [
  "The cat sat on the mat",
  "The dog played in the park",
  "Cat and dogs are great pets"
]
vectorizer = TfidfVectorizer()
tfidf = vectorizer.fit_transform(docs)
similarity = cosine_similarity(tfidf)
print("Cosine Similarity Matrix")
print(similarity)

5. Inverted Index
docs = [
  "cat sat on mat",
  "dog played in park",
  "cat and dog"
]
index = {}
for i, doc in enumerate(docs, 1):
  for word in doc.split():
    if word not in index:
      index[word] = []
    index[word].append(i)
print("Inverted Index")
for word, doclist in index.items():
  print(word, ":", doclist)

6. Sentiment Analysis using TextBlob
from textblob import TextBlob
text = input("Enter sentence: ")
analysis = TextBlob(text)
print("Polarity:", analysis.sentiment.polarity)
print("Subjectivity:", analysis.sentiment.subjectivity)
if analysis.sentiment.polarity > 0:
  print("Positive")
elif analysis.sentiment.polarity < 0:
  print("Negative")
else:
  print("Neutral")

7. Sentiment Analysis using VADER
from vaderSentiment.vaderSentiment import SentimentIntensityAnalyzer
analyzer = SentimentIntensityAnalyzer()
text = input("Enter sentence: ")
score = analyzer.polarity_scores(text)
print(score)
if score["compound"] >= 0.05:
  print("Positive")
elif score["compound"] <= -0.05:
  print("Negative")
else:
  print("Neutral")
